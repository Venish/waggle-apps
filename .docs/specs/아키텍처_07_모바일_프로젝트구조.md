## 1. 네이티브 프로젝트 위치 결정

### 1.1 결정: `packages/native-walk` 독립 패키지

산책 네이티브 모듈은 **`packages/native-walk/`** 에 독립 패키지(`@waggle/native-walk`)로 분리한다.

**이유:**
- `apps/mobile-app/android/`, `ios/` 안에 포함하면 monorepo 다른 앱(미래의 TV앱, 워치앱 등)에서 재사용 불가
- 독립 패키지로 분리하면 `apps/poc-walk-ios/`, `apps/poc-walk-android/` PoC 앱에서도 동일 인터페이스 검증 가능
- 버그 수정을 네이티브 패키지만 릴리즈하여 JS bundle OTA(Shorebird)와 분리 추적 가능
- `react-native-vision-camera`, `react-native-mmkv` 등 우수 RN 라이브러리와 동일한 구조

**모노레포 내 native 패키지 패턴 (참고: react-native-builder-bob):**
```
packages/native-walk/
├── ios/          ← Swift 소스 (XCFramework 아님, 소스 직접 포함)
├── android/      ← Kotlin 소스 (AAR 아님, 소스 직접 포함)
├── src/          ← TypeScript Codegen 스펙 + JS 래퍼
└── package.json  ← peerDependencies: react-native
```

`apps/mobile-app/package.json`에서 `"@waggle/native-walk": "workspace:*"` 로 참조.

### 1.2 PoC 앱 위치: `apps/poc-walk/`

PoC는 순수 네이티브 앱(RN 없음)으로, iOS와 Android 각각 독립 프로젝트:

```
apps/poc-walk/
├── ios/      ← Xcode 프로젝트 (Swift)
└── android/  ← Android Studio 프로젝트 (Kotlin)
```

PoC 통과 후 이 디렉터리는 `archived/poc-walk/`로 이동하거나 그대로 유지(레퍼런스용).

### 3.2 Codegen 스펙 — TurboModule

```typescript
// packages/native-walk/src/NativeWalkTracker.ts
import type { TurboModule } from 'react-native';
import { TurboModuleRegistry } from 'react-native';

export type WalkState = 'idle' | 'tracking' | 'paused' | 'completed';

export type WalkConfig = {
  minDistanceMeters: number;   // 기본: 5
  minTimeSeconds: number;      // 기본: 3
  kalmanQ: number;             // Kalman 프로세스 노이즈 (기본: 3)
  kalmanR: number;             // Kalman 측정 노이즈 (기본: 10)
  dpEpsilonMeters: number;     // Douglas-Peucker 임계값 (기본: 2)
};

export type WalkStats = {
  distanceMeters: number;
  durationSeconds: number;
  speedMps: number;            // 현재 속도 (m/s)
  pointCount: number;
};

export type WalkSummary = {
  sessionId: string;
  startedAt: string;           // ISO 8601
  endedAt: string;
  distanceMeters: number;
  durationSeconds: number;
  avgSpeedMps: number;
  pointCount: number;
  encodedPolyline: string;     // Google Encoded Polyline
  boundingBox: {
    northEast: { lat: number; lng: number };
    southWest: { lat: number; lng: number };
  };
};

export interface Spec extends TurboModule {
  // 세션 제어
  start(dogId: string, config: WalkConfig): Promise<string>;    // → sessionId
  pause(): Promise<void>;
  resume(): Promise<void>;
  stop(): Promise<WalkSummary>;
  getCurrentSession(): Promise<{ sessionId: string; stats: WalkStats; state: WalkState } | null>;

  // 이벤트 구독 (RN EventEmitter)
  addListener(eventName: string): void;
  removeListeners(count: number): void;
}

export default TurboModuleRegistry.getEnforcing<Spec>('WalkTracker');
```

```swift
// packages/native-walk/ios/WalkTrackingManager.swift
import CoreLocation
import CoreData

@objc public class WalkTrackingManager: NSObject, CLLocationManagerDelegate {
    
    private let locationManager = CLLocationManager()
    private let locationFilter: LocationFilter
    private let sessionStore: WalkSessionStore
    private let sensorFusion: SensorFusion
    
    private(set) var currentSessionId: String?
    private(set) var state: WalkState = .idle
    
    // 통계 (3초마다 JS에 전달)
    private(set) var stats = WalkStats()
    
    // 이벤트 콜백 (TurboModule에서 주입)
    var onStatsUpdate: ((WalkStats) -> Void)?
    var onStateChange: ((WalkState) -> Void)?
    
    private let statsThrottleInterval: TimeInterval = 3.0
    private var lastStatsEmitTime: Date = .distantPast
    private var rawPoints: [CLLocation] = []
    
    public override init() {
        self.locationFilter = LocationFilter()
        self.sessionStore = WalkSessionStore()
        self.sensorFusion = SensorFusion()
        super.init()
        setupLocationManager()
    }
    
    private func setupLocationManager() {
        locationManager.delegate = self
        locationManager.desiredAccuracy = kCLLocationAccuracyBest
        locationManager.distanceFilter = 5       // 5m 미만 업데이트 무시
        locationManager.pausesLocationUpdatesAutomatically = false
        locationManager.allowsBackgroundLocationUpdates = true    // Info.plist: UIBackgroundModes = location
        locationManager.showsBackgroundLocationIndicator = true
    }
    
    // MARK: - Public API
    
    public func start(dogId: String, config: WalkConfig) throws -> String {
        guard state == .idle else { throw WalkError.invalidState }
        let sessionId = UUID().uuidString
        currentSessionId = sessionId
        rawPoints = []
        stats = WalkStats()
        sessionStore.createSession(id: sessionId, dogId: dogId)
        locationManager.requestWhenInUseAuthorization()
        locationManager.startUpdatingLocation()
        sensorFusion.start()
        setState(.tracking)
        return sessionId
    }
    
    public func pause() throws {
        guard state == .tracking else { throw WalkError.invalidState }
        locationManager.stopUpdatingLocation()
        sensorFusion.pause()
        setState(.paused)
    }
    
    public func resume() throws {
        guard state == .paused else { throw WalkError.invalidState }
        locationManager.startUpdatingLocation()
        sensorFusion.resume()
        setState(.tracking)
    }
    
    public func stop() throws -> WalkSummary {
        guard state == .tracking || state == .paused else { throw WalkError.invalidState }
        locationManager.stopUpdatingLocation()
        sensorFusion.stop()
        
        let filtered = locationFilter.douglasPeucker(points: rawPoints, epsilon: 2.0)
        let summary = sessionStore.finalizeSession(
            id: currentSessionId!,
            filteredPoints: filtered,
            stats: stats
        )
        setState(.idle)
        currentSessionId = nil
        return summary
    }
    
    // MARK: - CLLocationManagerDelegate
    
    public func locationManager(_ manager: CLLocationManager, didUpdateLocations locations: [CLLocation]) {
        guard state == .tracking else { return }
        
        for location in locations {
            // 정확도 필터 (50m 이상이면 버림)
            guard location.horizontalAccuracy > 0,
                  location.horizontalAccuracy <= 50 else { continue }
            
            // 센서 퓨전: 정지 상태 감지 시 스킵
            guard !sensorFusion.isStationary else { continue }
            
            // Kalman 필터 적용
            let filtered = locationFilter.kalmanFilter(location: location)
            rawPoints.append(filtered)
            
            // 통계 업데이트
            updateStats(newLocation: filtered)
            
            // 네이티브 DB에 즉시 저장 (서버 업로드와 무관하게)
            sessionStore.appendPoint(sessionId: currentSessionId!, location: filtered)
        }
        
        // JS 통보 — 3초 throttle
        let now = Date()
        if now.timeIntervalSince(lastStatsEmitTime) >= statsThrottleInterval {
            onStatsUpdate?(stats)
            lastStatsEmitTime = now
        }
    }
    
    private func updateStats(newLocation: CLLocation) {
        if let last = rawPoints.dropLast().last {
            let delta = newLocation.distance(from: last)
            stats.distanceMeters += delta
            stats.speedMps = newLocation.speed > 0 ? newLocation.speed : 0
        }
        stats.durationSeconds = Int(Date().timeIntervalSince(sessionStore.sessionStartTime(id: currentSessionId!) ?? Date()))
        stats.pointCount = rawPoints.count
    }
    
    private func setState(_ newState: WalkState) {
        state = newState
        onStateChange?(newState)
    }
}

// MARK: - Supporting Types

@objc public enum WalkState: Int { case idle, tracking, paused, completed }

public struct WalkStats {
    var distanceMeters: Double = 0
    var durationSeconds: Int = 0
    var speedMps: Double = 0
    var pointCount: Int = 0
}

public struct WalkConfig {
    let minDistanceMeters: Double
    let minTimeSeconds: Double
    let kalmanQ: Double
    let kalmanR: Double
    let dpEpsilonMeters: Double
}

public enum WalkError: Error {
    case invalidState
    case permissionDenied
    case sessionNotFound
}
```

### 6.1 `WalkTrackingService.kt` (ForegroundService)

```kotlin
// packages/native-walk/android/src/main/java/com/waggle/nativewalk/WalkTrackingService.kt
package com.waggle.nativewalk

import android.app.*
import android.content.Intent
import android.os.*
import androidx.core.app.NotificationCompat
import com.google.android.gms.location.*
import kotlinx.coroutines.*

class WalkTrackingService : Service() {

    companion object {
        const val CHANNEL_ID = "waggle_walk"
        const val NOTIFICATION_ID = 1001
        const val ACTION_START  = "ACTION_START"
        const val ACTION_PAUSE  = "ACTION_PAUSE"
        const val ACTION_RESUME = "ACTION_RESUME"
        const val ACTION_STOP   = "ACTION_STOP"

        // 다른 컴포넌트에서 현재 상태에 접근 (싱글턴 접근자)
        var instance: WalkTrackingService? = null
            private set
    }

    private lateinit var fusedLocationSource: FusedLocationSource
    private lateinit var locationFilter: LocationFilter
    private lateinit var sessionStore: WalkSessionStore
    private lateinit var sensorFusion: SensorFusion

    private val scope = CoroutineScope(Dispatchers.Default + SupervisorJob())
    private var currentSessionId: String? = null
    private var state = WalkState.IDLE

    // 통계 콜백 (TurboModule에서 등록)
    var onStatsUpdate: ((WalkStats) -> Unit)? = null
    var onStateChange: ((WalkState) -> Unit)? = null

    override fun onCreate() {
        super.onCreate()
        instance = this
        fusedLocationSource = FusedLocationSource(this, ::onLocationUpdate)
        locationFilter = LocationFilter()
        sessionStore = WalkSessionStore(this)
        sensorFusion = SensorFusion(this)
        createNotificationChannel()
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        when (intent?.action) {
            ACTION_START  -> intent.getStringExtra("dogId")?.let { startWalk(it) }
            ACTION_PAUSE  -> pauseWalk()
            ACTION_RESUME -> resumeWalk()
            ACTION_STOP   -> stopWalk()
        }
        return START_STICKY    // OS kill 후 재시작
    }

    fun startWalk(dogId: String): String {
        val sessionId = java.util.UUID.randomUUID().toString()
        currentSessionId = sessionId
        sessionStore.createSession(sessionId, dogId)
        startForeground(NOTIFICATION_ID, buildNotification("산책 준비 중..."))
        fusedLocationSource.start()
        sensorFusion.start()
        setState(WalkState.TRACKING)
        startStatsLoop()
        return sessionId
    }

    fun pauseWalk() {
        fusedLocationSource.stop()
        sensorFusion.pause()
        setState(WalkState.PAUSED)
    }

    fun resumeWalk() {
        fusedLocationSource.start()
        sensorFusion.resume()
        setState(WalkState.TRACKING)
    }

    fun stopWalk(): WalkSummary? {
        fusedLocationSource.stop()
        sensorFusion.stop()
        scope.coroutineContext.cancelChildren()
        val summary = currentSessionId?.let { sessionStore.finalizeSession(it) }
        setState(WalkState.IDLE)
        currentSessionId = null
        stopForeground(STOP_FOREGROUND_REMOVE)
        stopSelf()
        return summary
    }

    fun getCurrentStats(): WalkStats? = currentSessionId?.let { sessionStore.getStats(it) }

    private fun onLocationUpdate(location: android.location.Location) {
        if (state != WalkState.TRACKING) return
        if (sensorFusion.isStationary) return
        if (location.accuracy > 50f) return    // 50m 이상 정확도 버림

        val filtered = locationFilter.kalmanFilter(location)
        sessionStore.appendPoint(currentSessionId!!, filtered)
        updateNotification()
    }

    private fun startStatsLoop() {
        scope.launch {
            while (isActive) {
                delay(3000L)   // 3초 throttle
                val stats = getCurrentStats() ?: continue
                withContext(Dispatchers.Main) { onStatsUpdate?.invoke(stats) }
                updateNotification()
            }
        }
    }

    private fun updateNotification() {
        val stats = getCurrentStats() ?: return
        val km = stats.distanceMeters / 1000.0
        val nm = getSystemService(NOTIFICATION_SERVICE) as NotificationManager
        nm.notify(NOTIFICATION_ID, buildNotification("🐾 산책 중  %.1fkm".format(km)))
    }

    private fun buildNotification(text: String): Notification {
        return NotificationCompat.Builder(this, CHANNEL_ID)
            .setContentTitle("Waggle 산책 중")
            .setContentText(text)
            .setSmallIcon(android.R.drawable.ic_menu_mylocation)
            .setOngoing(true)
            .setPriority(NotificationCompat.PRIORITY_LOW)
            .build()
    }

    private fun createNotificationChannel() {
        val channel = NotificationChannel(CHANNEL_ID, "산책 추적", NotificationManager.IMPORTANCE_LOW)
        channel.description = "산책 거리 및 경로를 기록합니다"
        (getSystemService(NOTIFICATION_SERVICE) as NotificationManager).createNotificationChannel(channel)
    }

    private fun setState(newState: WalkState) {
        state = newState
        onStateChange?.invoke(newState)
    }

    override fun onBind(intent: Intent?) = null

    override fun onDestroy() {
        super.onDestroy()
        instance = null
        scope.cancel()
    }
}

enum class WalkState { IDLE, TRACKING, PAUSED, COMPLETED }
data class WalkStats(val distanceMeters: Double = 0.0, val durationSeconds: Int = 0, val speedMps: Double = 0.0, val pointCount: Int = 0)
```

### 6.2 `WalkTrackerModule.kt` (TurboModule)

```kotlin
// packages/native-walk/android/src/main/java/com/waggle/nativewalk/WalkTrackerModule.kt
package com.waggle.nativewalk

import android.content.Intent
import com.facebook.react.bridge.*
import com.facebook.react.modules.core.DeviceEventManagerModule

class WalkTrackerModule(reactContext: ReactApplicationContext) :
    NativeWalkTrackerSpec(reactContext) {

    override fun getName() = "WalkTracker"

    init {
        // WalkTrackingService 콜백 등록
        // (Service가 아직 실행 안 됐을 수도 있으므로 Service 내부에서 등록)
    }

    override fun start(dogId: String, config: ReadableMap, promise: Promise) {
        val intent = Intent(reactApplicationContext, WalkTrackingService::class.java).apply {
            action = WalkTrackingService.ACTION_START
            putExtra("dogId", dogId)
        }
        reactApplicationContext.startForegroundService(intent)

        // Service 인스턴스가 생성될 때까지 짧게 대기 후 sessionId 반환
        android.os.Handler(android.os.Looper.getMainLooper()).postDelayed({
            val service = WalkTrackingService.instance
            if (service != null) {
                // 콜백 등록
                service.onStatsUpdate = { stats -> sendEvent("stats", stats.toWritableMap()) }
                service.onStateChange = { state -> sendEvent("state", WritableNativeMap().apply { putString("state", state.name.lowercase()) }) }
                // Service가 startWalk를 onStartCommand에서 이미 호출했으므로 sessionId 반환
                promise.resolve(service.currentSessionId ?: "unknown")
            } else {
                promise.reject("service_error", "WalkTrackingService not started")
            }
        }, 200)
    }

    override fun stop(promise: Promise) {
        val service = WalkTrackingService.instance
        if (service == null) { promise.reject("not_running", "Walk not in progress"); return }
        val summary = service.stopWalk()
        if (summary != null) promise.resolve(summary.toWritableMap())
        else promise.reject("stop_failed", "Failed to get summary")
    }

    override fun pause(promise: Promise) {
        WalkTrackingService.instance?.pauseWalk()
        promise.resolve(null)
    }

    override fun resume(promise: Promise) {
        WalkTrackingService.instance?.resumeWalk()
        promise.resolve(null)
    }

    override fun getCurrentSession(promise: Promise) {
        val service = WalkTrackingService.instance
        if (service == null) { promise.resolve(null); return }
        val stats = service.getCurrentStats()
        if (stats == null) { promise.resolve(null); return }
        promise.resolve(WritableNativeMap().apply {
            putString("sessionId", service.currentSessionId)
            putString("state", service.state.name.lowercase())
            putMap("stats", stats.toWritableMap())
        })
    }

    override fun addListener(eventName: String) {}
    override fun removeListeners(count: Double) {}

    private fun sendEvent(name: String, params: WritableMap) {
        reactApplicationContext
            .getJSModule(DeviceEventManagerModule.RCTDeviceEventEmitter::class.java)
            .emit(name, params)
    }

    private fun WalkStats.toWritableMap() = WritableNativeMap().apply {
        putDouble("distanceMeters", distanceMeters)
        putInt("durationSeconds", durationSeconds)
        putDouble("speedMps", speedMps)
        putInt("pointCount", pointCount)
    }
}
```
